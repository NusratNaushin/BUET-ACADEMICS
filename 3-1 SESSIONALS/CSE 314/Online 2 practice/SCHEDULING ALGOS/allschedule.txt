void scheduler(void)
{

  struct proc *p;
  struct cpu *c = mycpu();
  c->proc = 0;

  for (;;)
  {

    int found = 0;


    struct proc *highPriority = 0;
    for(p = proc; p < &proc[NPROC]; p++) {
      acquire(&p->lock);
      if(p->state == RUNNABLE) {
        if(highPriority == 0 || (p->priority  > highPriority->priority)){
         highPriority = p;
        }
      }
      release(&p->lock);
    }

//     //p=shortest_job;
    if (highPriority != 0){

      acquire(&highPriority->lock);

      if(highPriority->state == RUNNABLE){
        highPriority->state = RUNNING;
        c->proc = highPriority;
        swtch(&c->context, &highPriority->context);

        c->proc = 0;
        found = 1;
      }

      release(&highPriority->lock);

    }
    if (found == 0)
    {
      // nothing to run; stop running on this core until an interrupt.
      intr_on();
      asm volatile("wfi");
    }
  }
}



preemptive sjf

void
scheduler(void)
{
  struct proc *p;
  struct cpu *c = mycpu();
  c->proc = 0;

  for(;;){
    intr_on();
    struct proc *shortest_job = 0;

    for(p = proc; p < &proc[NPROC]; p++){
      acquire(&p->lock);
      if(p->state == RUNNABLE){
        if(shortest_job == 0 || p->remaining_time < shortest_job->remaining_time){
          if(shortest_job)
            release(&shortest_job->lock);
          shortest_job = p;
        } else {
          release(&p->lock);
        }
      } else {
        release(&p->lock);
      }
    }

    if(shortest_job != 0){
      // Run the shortest job
      shortest_job->state = RUNNING;
      c->proc = shortest_job;
      swtch(&c->context, &shortest_job->context);
      c->proc = 0;
      release(&shortest_job->lock);
    } else {
      asm volatile("wfi");
    }
  }
}


round robin

void
scheduler(void)
{
  struct proc *p;
  struct cpu *c = mycpu();
  c->proc = 0;

  for(;;){
    // enable interrupts
    intr_on();

    for(p = proc; p < &proc[NPROC]; p++){
      acquire(&p->lock);
      if(p->state == RUNNABLE){
        // Assign process to CPU and switch to it
        p->state = RUNNING;
        c->proc = p;
        swtch(&c->context, &p->context);
        // Process yielded
        c->proc = 0;
      }
      release(&p->lock);
    }

    // If no RUNNABLE process was found, sleep
    asm volatile("wfi");
  }
}
